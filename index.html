Database Setup Required

Database setup needed: The rooms table or RLS policies have not been created in your Supabase project yet.

Follow these 3 quick steps to enable multiplayer gameplay:

  1. Click Copy SQL Script below.
  2. Open your project's Supabase SQL Editor ↗.
  3. Paste the script, click Run, then return here!
-- 1. Create the rooms table
create table if not exists public.rooms (
  id uuid primary key default gen_random_uuid(),
  code text not null unique,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  status text not null default 'waiting',
  turn text not null default 'X',
  board jsonb not null default '["", "", "", "", "", "", "", "", ""]'::jsonb,
  winner text default null,
  winning_combo jsonb default null,
  player_x jsonb not null default '{}'::jsonb,
  player_o jsonb default null,
  draws integer not null default 0,
  abandoned_by text default null
);

-- 2. Ensure full row data is broadcast in update payloads
alter table public.rooms replica identity full;

-- 3. Enable Row Level Security (RLS)
alter table public.rooms enable row level security;

-- 4. Create permissive policies for open two-player sessions
drop policy if exists "Allow public read on rooms" on public.rooms;
create policy "Allow public read on rooms"
  on public.rooms for select
  using (true);

drop policy if exists "Allow public insert on rooms" on public.rooms;
create policy "Allow public insert on rooms"
  on public.rooms for insert
  with check (true);

drop policy if exists "Allow public update on rooms" on public.rooms;
create policy "Allow public update on rooms"
  on public.rooms for update
  using (true)
  with check (true);

-- 5. Enable Realtime broadcasting on the rooms table
do $$
begin
  if not exists (
    select 1 from pg_publication_tables 
    where pubname = 'supabase_realtime' and schemaname = 'public' and tablename = 'rooms'
  ) then
    alter publication supabase_realtime add table public.rooms;
  end if;
end $$;
